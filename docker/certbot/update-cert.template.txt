#!/bin/sh

# Update certificate
certbot renew --webroot \
    --webroot-path=/var/www/html \
    --email "${CERTBOT_EMAIL}" \
    --agree-tos \
    --no-eff-email \
    --non-interactive \
    --domain "${CERTBOT_DOMAIN}" \
    ${CERTBOT_OPTIONS}

# Reload nginx to pick up new certificates
nginx -s reload
